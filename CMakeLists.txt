cmake_minimum_required(VERSION 3.24)
project(Scattuccino VERSION 0.0.1 LANGUAGES NONE)
option(SCATTUCCINO_BUILD_CORE "Build the C++ module scattuccino._core" ON)
option(SCATTUCCINO_ENABLE_CUDA "Build the CUDA backend (stage S5b)" OFF)
if(NOT SCATTUCCINO_BUILD_CORE)
  message(STATUS "Scattuccino: core disabled, pure Python reference only")
  return()
endif()
enable_language(CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(scattuccino_core STATIC cpp/src/core.cpp)
target_include_directories(scattuccino_core PUBLIC cpp/include)
set_target_properties(scattuccino_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(scattuccino_core PRIVATE /W4 /fp:precise)
else()
  # no FMA contraction: same rounding as NumPy for the arithmetic (not for log/sin/cos)
  target_compile_options(scattuccino_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(scattuccino_core PUBLIC OpenMP::OpenMP_CXX)
endif()
if(SKBUILD)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/python/bindings.cpp)
  target_link_libraries(_core PRIVATE scattuccino_core)
  install(TARGETS _core LIBRARY DESTINATION scattuccino)
endif()
