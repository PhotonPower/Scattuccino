cmake_minimum_required(VERSION 3.24)
project(Scattuccino VERSION 0.0.1 LANGUAGES NONE)
option(SCATTUCCINO_BUILD_CORE "Build the C++ module scattuccino._core" ON)
option(SCATTUCCINO_ENABLE_CUDA "Build the CUDA backend (stage S5b)" OFF)
if(NOT SCATTUCCINO_BUILD_CORE)
  message(STATUS "Scattuccino: core disabled, pure Python reference only")
  return()
endif()
enable_language(CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(scattuccino_core STATIC cpp/src/core.cpp cpp/src/tets.cpp)
target_include_directories(scattuccino_core PUBLIC cpp/include)
set_target_properties(scattuccino_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(scattuccino_core PRIVATE /W4 /fp:precise)
else()
  # no FMA contraction: same rounding as NumPy for the arithmetic (not for log/sin/cos)
  target_compile_options(scattuccino_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(scattuccino_core PUBLIC OpenMP::OpenMP_CXX)
endif()
option(SCATTUCCINO_STATIC_LIBSTDCXX "Link libstdc++/libgcc statically (conda Pythons)" OFF)
if(SCATTUCCINO_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 89)  # RTX 6000 Ada
  endif()
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  find_package(CUDAToolkit 12.6 REQUIRED)
  add_library(scattuccino_cuda STATIC cpp/src/core_cuda.cu)
  target_include_directories(scattuccino_cuda PUBLIC cpp/include)
  # same rounding as the CPU core: no FMA contraction, IEEE division and square root
  target_compile_options(scattuccino_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true
                         --expt-extended-lambda)
  target_link_libraries(scattuccino_cuda PUBLIC CUDA::cudart_static scattuccino_core)
  set_target_properties(scattuccino_cuda PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()

if(SKBUILD)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/python/bindings.cpp)
  target_link_libraries(_core PRIVATE scattuccino_core)
  if(SCATTUCCINO_ENABLE_CUDA)
    target_link_libraries(_core PRIVATE scattuccino_cuda)
    target_compile_definitions(_core PRIVATE SCATTUCCINO_HAVE_CUDA)
  endif()
  if(SCATTUCCINO_STATIC_LIBSTDCXX AND NOT MSVC)
    target_link_options(_core PRIVATE -static-libstdc++ -static-libgcc)
  endif()
  install(TARGETS _core LIBRARY DESTINATION scattuccino)
endif()
